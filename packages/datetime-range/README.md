# @smartput/datetime-range

> Full instants at both ends, holidays optional.

The widest of the four. Its `./holiday` subpath carries the same
opt-in cost as `datetime`'s, for the same reason and behind the same size
row.

## Setup

```sh
npm add @smartput/datetime-range
```

## Example

```ts
import { composeLocale, createEngine } from "@smartput/core";
import { english } from "@smartput/core/locale/en";
import { BUILTIN_KINDS } from "@smartput/kinds";
import BUILTIN_EN from "@smartput/kinds/locale/en";
import { datetime } from "@smartput/datetime";
import datetimeEn from "@smartput/datetime/locale/en";
import { date } from "@smartput/date";
import { time } from "@smartput/time";
import { datetimeRange } from "@smartput/datetime-range";

const engine = createEngine({
  locales: [composeLocale(english, [...BUILTIN_EN, datetimeEn])],
  kinds: [...BUILTIN_KINDS, datetime, date, time, datetimeRange],
  timeZone: "Europe/Kyiv",
});

engine.evaluate("yesterday morning").formatted        // "2026-01-14 06:00 → 2026-01-14 12:00 UTC"
engine.evaluate("tomorrow night").formatted           // "2026-01-16 22:00 → 2026-01-17 06:00 UTC"
engine.evaluate("from today until friday").formatted  // "2026-01-15 00:00 → 2026-01-16 00:00 UTC"
```
