# @smartput/time

> A clock time, with no date attached.

Nanoseconds since midnight, wrapped as a kind. The counterpart to
[`@smartput/date`](../date/README.md): together they are what
[`@smartput/datetime`](../datetime/README.md) splits into.

## Setup

```sh
npm add @smartput/time
```

## Example

```ts
import { composeLocale, createEngine } from "@smartput/core";
import { english } from "@smartput/core/locale/en";
import { BUILTIN_KINDS } from "@smartput/kinds";
import BUILTIN_EN from "@smartput/kinds/locale/en";
import { datetime } from "@smartput/datetime";
import datetimeEn from "@smartput/datetime/locale/en";
import { time } from "@smartput/time";

const engine = createEngine({
  locales: [composeLocale(english, [...BUILTIN_EN, datetimeEn])],
  kinds: [...BUILTIN_KINDS, datetime, time],
  timeZone: "Europe/Kyiv",
});

engine.evaluate("3pm").formatted    // "2026-01-15 15:00 UTC"
engine.evaluate("09:30").formatted  // "2026-01-15 09:30 UTC"
engine.evaluate("noon").formatted   // "2026-01-15 12:00 UTC"
```
