# @smartput/date-range

> `last week`, `March 3–7`, `between May and June`.

Date endpoints plus the phrase table that resolves a named span to two of them.

## Setup

```sh
npm add @smartput/date-range
```

## Example

```ts
import { composeLocale, createEngine } from "@smartput/core";
import { english } from "@smartput/core/locale/en";
import { BUILTIN_KINDS } from "@smartput/kinds";
import BUILTIN_EN from "@smartput/kinds/locale/en";
import { datetime } from "@smartput/datetime";
import datetimeEn from "@smartput/datetime/locale/en";
import { date } from "@smartput/date";
import { dateRange } from "@smartput/date-range";

const engine = createEngine({
  locales: [composeLocale(english, [...BUILTIN_EN, datetimeEn])],
  kinds: [...BUILTIN_KINDS, datetime, date, dateRange],
  timeZone: "Europe/Kyiv",
});

engine.evaluate("whole week").formatted         // "2026-01-12 → 2026-01-18"
engine.evaluate("today to friday").formatted    // "2026-01-15 → 2026-01-16"
engine.evaluate("next month").formatted         // "2026-02-01 → 2026-02-28"
engine.evaluate("whole week + 1 wk").formatted  // "2026-01-19 → 2026-01-25"
```
